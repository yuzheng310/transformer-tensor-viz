% Placeholder only. Replace with exact supplied values when cell_values=on.
\newcommand{\CellValue}[2]{$#1#2$}

% Neutral T x d input grid. Head identity does not exist before projection.
\newcommand{\InputFace}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \r in {0,...,5} {
      \foreach \c in {0,...,5} {
        \filldraw[fill=pastelNeutral,draw=outlineBlue!72,line width=0.42pt]
          ({\c*0.43},{-\r*0.36}) rectangle ++(0.43,-0.36);
        \ifshowcellvalues
          \node[font=\tiny,text=black!66] at
            ({\c*0.43+0.215},{-\r*0.36-0.18}) {\CellValue{\r}{\c}};
        \fi
      }
    }
    \draw[outlineBlue,line width=0.65pt] (0,0) rectangle (2.58,-2.16);
    \node[font=\small,anchor=north] at (1.29,-2.42) {#3};
    \node[font=\scriptsize,text=black!56,anchor=north] at (1.29,-2.82) {#4};
  \end{scope}%
}

% Continuous T x d grid for d=6, h=3, d_h=2. Each two-column
% band has the same identity and color as exactly one output head.
\newcommand{\TensorFace}[4]{%
  \begin{scope}[shift={(#1,#2)}]
    \foreach \r in {0,...,5} {
      \foreach \c in {0,...,5} {
        \pgfmathtruncatemacro{\group}{floor(\c/2)}
        \ifcase\group\def\cellcolor{pastelGreen}\or
          \def\cellcolor{pastelOrange}\or
          \def\cellcolor{pastelBlue}\fi
        \filldraw[fill=\cellcolor,draw=outlineBlue!72,line width=0.42pt]
          ({\c*0.43},{-\r*0.36}) rectangle ++(0.43,-0.36);
        \ifshowcellvalues
          \node[font=\tiny,text=black!66] at
            ({\c*0.43+0.215},{-\r*0.36-0.18}) {\CellValue{\r}{\c}};
        \fi
      }
    }
    \draw[outlineBlue,line width=0.65pt] (0,0) rectangle (2.58,-2.16);
    \node[font=\small,anchor=north] at (1.29,-2.42) {#3};
    \node[font=\scriptsize,text=black!56,anchor=north] at (1.29,-2.82) {#4};
  \end{scope}%
}

% h x (T x d_h): the front face is T x d_h and offset sheets encode heads.
\newcommand{\HeadStack}[4]{%
  \foreach \s in {2,1,0} {
    \begin{scope}[shift={({#1+0.20*\s},{#2+0.16*\s})}]
      \foreach \r in {0,...,5} {
        \foreach \c in {0,...,1} {
          \pgfmathtruncatemacro{\group}{\s}
          \ifcase\group\def\cellcolor{pastelGreen}\or
            \def\cellcolor{pastelOrange}\or
            \def\cellcolor{pastelBlue}\fi
          \filldraw[fill=\cellcolor,draw=outlineBlue!72,line width=0.42pt]
            ({\c*0.43},{-\r*0.36}) rectangle ++(0.43,-0.36);
          % Offset back sheets encode depth only. If every head needs readable
          % values, replace the stack with separate panels.
          \ifshowcellvalues
            \ifnum\s=0\relax
              \node[font=\tiny,text=black!66] at
                ({\c*0.43+0.215},{-\r*0.36-0.18}) {\CellValue{\r}{\c}};
            \fi
          \fi
        }
      }
      \draw[outlineBlue,line width=0.65pt] (0,0) rectangle (0.86,-2.16);
    \end{scope}
  }
  \node[font=\small,anchor=north] at ({#1+0.43},{#2-2.42}) {#3};
  \node[font=\scriptsize,text=black!56,anchor=north]
    at ({#1+0.43},{#2-2.82}) {#4};
}
